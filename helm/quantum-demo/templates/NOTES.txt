**After Helm Command Completion:**

*   **Copy Outputs (on Success):**
    export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "job-name={{ include "quantum-demo.fullname" . }}" -o jsonpath="{.items[0].metadata.name}")
    if [ -z "$POD_NAME" ]; then
      echo "Pod not found."
    else
      POD_STATUS=$(kubectl get pod $POD_NAME -n {{ .Release.Namespace }} -o jsonpath='{.status.phase}')
      if [ "$POD_STATUS" == "Succeeded" ]; then
        mkdir -p ./kubecon_outputs
        kubectl --namespace {{ .Release.Namespace }} cp ${POD_NAME}:/app/outputs ./kubecon_outputs/
        echo "Output files copied to ./kubecon_outputs/outputs"
      else
        echo "Pod $POD_NAME did not Succeed. Status: $POD_STATUS"
      fi
    fi
